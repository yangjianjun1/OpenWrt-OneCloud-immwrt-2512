setenv condev "console=ttyAML0,115200n8 no_console_suspend consoleblank=0"
setenv kernel_loadaddr "0x00208000"
setenv dtb_loadaddr "0x21800000"
setenv dtb_name "dtb"
setenv boot_start "bootm ${kernel_loadaddr} - ${dtb_loadaddr}"

if usb start; then
    if fatload usb 0 ${kernel_loadaddr} uImage &&
       fatload usb 0 ${dtb_loadaddr} ${dtb_name} &&
       setenv bootargs "root=/dev/sda2 rootwait ro ${condev} fsck.repair=yes net.ifnames=0"; then
        run boot_start
    fi
fi

if fatload mmc 1 ${kernel_loadaddr} uImage &&
   fatload mmc 1 ${dtb_loadaddr} ${dtb_name} &&
   setenv bootargs "root=/dev/mmcblk1p2 rootwait ro ${condev} fsck.repair=yes net.ifnames=0"; then
    run boot_start
fi
